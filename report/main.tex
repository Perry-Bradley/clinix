% --- "plain" page style: used on first page of every chapter, TOC, etc.

  \fancyhead[R]{\thepage}%

% Chapter title formatting - sizes follow the school template:
%   body text   = 12 pt  (set by documentclass)
%   subheading  = 14 pt  (\subsection)
%   heading     = 16 pt  (\section AND \chapter title body)

% Roman-numeral chapters (CHAPTER I, CHAPTER II, ...)
\renewcommand{\thechapter}{\Roman{chapter}}

% Chapter heading: "CHAPTER N" on its own line, then the title below
\titleformat{\chapter}[display]
  {\normalfont\fontsize{16}{19}\selectfont\bfseries\centering}
  {CHAPTER \thechapter}
  {0.6em}
  {}
\titlespacing*{\chapter}{0pt}{-20pt}{20pt}

% Section: 16 pt bold (heading)
\titleformat{\section}
  {\normalfont\fontsize{16}{19}\selectfont\bfseries}{\thesection}{0.6em}{}

% Subsection: 14 pt bold (subheading)
\titleformat{\subsection}
  {\normalfont\fontsize{14}{17}\selectfont\bfseries}{\thesubsection}{0.5em}{}

% Sub-subsection (used for the flat TOC sub-items): 12 pt bold (body weight)
\titleformat{\subsubsection}
  {\normalfont\fontsize{12}{15}\selectfont\bfseries}{\thesubsubsection}{0.4em}{}
